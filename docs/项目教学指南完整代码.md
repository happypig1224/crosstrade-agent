### 2.1 IntentType.java 意图类型枚举

> 文件路径：`crosstrade-common/src/main/java/com/crosstrade/common/enums/IntentType.java`

```java
package com.crosstrade.common.enums;

/**
 * 意图类型枚举
 * <p>
 * 定义系统支持的所有意图类型，与LLM输出、数据库存储保持一致。
 * 提供 fromValue() 方法，用于从字符串反序列化为枚举。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
public enum IntentType {

    PRODUCT_GUID("product_guide", "产品导购"),
    PRODUCT_INFO("product_info", "产品咨询"),
    QUOTE_NEGOTIATE("quote_negotiate", "报价谈判"),
    UNKNOWN("unknown", "未知意图");

    private final String value;
    private final String desc;

    IntentType(String value, String desc) {
        this.value = value;
        this.desc = desc;
    }

    public String getValue() {
        return value;
    }

    public String getDesc() {
        return desc;
    }

    /**
     * 从字符串值反序列化为枚举
     * LLM返回值可能不在预定义类型中，此时降级为UNKNOWN
     *
     * @param value 枚举字符串值
     * @return 对应的IntentType，未匹配时返回UNKNOWN
     */
    public static IntentType fromValue(String value) {
        if (value == null) {
            return UNKNOWN;
        }
        for (IntentType type : IntentType.values()) {
            if (type.value.equals(value)) {
                return type;
            }
        }
        return UNKNOWN;
    }
}
```

### 5.1 DfaSensitiveFilter.java DFA敏感词过滤器

> 文件路径：`crosstrade-common/src/main/java/com/crosstrade/common/util/DfaSensitiveFilter.java`

```java
package com.crosstrade.common.util;

import java.util.*;

/**
 * DFA（确定有限自动机）敏感词过滤器
 * <p>
 * 基于字典树算法的敏感词过滤，时间复杂度O(n)，与敏感词数量无关。
 * 优于正则匹配的O(n*m)复杂度。
 * <p>
 * DFA算法原理：将敏感词库构建成一棵多叉树，每个节点是一个Map，
 * key是字符，value是子节点Map。当某个路径的叶子节点标记为"结束"时，
 * 表示匹配到一个敏感词。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
public class DfaSensitiveFilter {

    /**
     * DFA字典树根节点
     */
    private final Map<Character, Object> sensitiveWordMap = new HashMap<>();

    /**
     * 敏感词结束标记Key
     */
    private static final String IS_END = "isEnd";

    /**
     * 初始化：从敏感词列表构建DFA树
     *
     * @param sensitiveWords 敏感词列表
     */
    public void initSensitiveWordMap(List<String> sensitiveWords) {
        if (sensitiveWords == null || sensitiveWords.isEmpty()) {
            return;
        }
        for (String word : sensitiveWords) {
            addWord(word);
        }
    }

    /**
     * 向DFA树中添加一个敏感词
     *
     * @param word 敏感词
     */
    @SuppressWarnings("unchecked")
    private void addWord(String word) {
        if (word == null || word.isEmpty()) {
            return;
        }
        Map<Character, Object> currentNode = sensitiveWordMap;
        for (char ch : word.toCharArray()) {
            // 如果当前字符不在树中，创建新节点
            if (!currentNode.containsKey(ch)) {
                currentNode.put(ch, new HashMap<Character, Object>());
            }
            // 移动到子节点
            currentNode = (Map<Character, Object>) currentNode.get(ch);
        }
        // 标记词结束
        currentNode.put(IS_END.charAt(0), true);
    }

    /**
     * 检测文本是否包含敏感词
     * <p>
     * 逐字符遍历文本，找到起始字符后沿DFA树向下匹配完整敏感词。
     *
     * @param text 待检测文本
     * @return true=包含敏感词，false=不包含
     */
    @SuppressWarnings("unchecked")
    public boolean containsSensitive(String text) {
        if (text == null || text.isEmpty()) {
            return false;
        }
        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);
            if (sensitiveWordMap.containsKey(ch)) {
                // 找到起始字符，沿DFA树向下匹配
                Map<Character, Object> currentNode = (Map<Character, Object>) sensitiveWordMap.get(ch);
                int j = i + 1;
                while (j < text.length()) {
                    char nextCh = text.charAt(j);
                    if (currentNode.containsKey(nextCh)) {
                        currentNode = (Map<Character, Object>) currentNode.get(nextCh);
                        // 检查是否到达敏感词结尾
                        if (currentNode.containsKey(IS_END.charAt(0))) {
                            return true; // 匹配到完整敏感词
                        }
                        j++;
                    } else {
                        break;
                    }
                }
                // 单字符敏感词的情况
                if (currentNode.containsKey(IS_END.charAt(0))) {
                    return true;
                }
            }
        }
        return false;
    }

    /**
     * 替换文本中的敏感词为指定字符
     *
     * @param text        待处理文本
     * @param replaceChar 替换字符（如*）
     * @return 替换后的文本
     */
    @SuppressWarnings("unchecked")
    public String filter(String text, char replaceChar) {
        if (text == null || text.isEmpty()) {
            return text;
        }
        char[] chars = text.toCharArray();
        for (int i = 0; i < chars.length; i++) {
            char ch = chars[i];
            if (sensitiveWordMap.containsKey(ch)) {
                Map<Character, Object> currentNode = (Map<Character, Object>) sensitiveWordMap.get(ch);
                int j = i + 1;
                int endIdx = -1;
                while (j < chars.length) {
                    char nextCh = chars[j];
                    if (currentNode.containsKey(nextCh)) {
                        currentNode = (Map<Character, Object>) currentNode.get(nextCh);
                        if (currentNode.containsKey(IS_END.charAt(0))) {
                            endIdx = j;
                        }
                        j++;
                    } else {
                        break;
                    }
                }
                // 单字符敏感词
                if (currentNode.containsKey(IS_END.charAt(0))) {
                    endIdx = i;
                }
                // 替换敏感词
                if (endIdx != -1) {
                    for (int k = i; k <= endIdx; k++) {
                        chars[k] = replaceChar;
                    }
                    i = endIdx;
                }
            }
        }
        return new String(chars);
    }
}
```

### 5.2 PromptInjectionDetector.java Prompt注入检测器

> 文件路径：`crosstrade-common/src/main/java/com/crosstrade/common/util/PromptInjectionDetector.java`

```java
package com.crosstrade.common.util;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

/**
 * Prompt注入检测器
 * <p>
 * 基于正则表达式检测常见的Prompt注入攻击模式。
 * 包括：忽略指令、角色扮演、系统提示泄露、规则覆盖等。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
public class PromptInjectionDetector {

    /**
     * 常见Prompt注入攻击正则模式列表
     */
    private final List<Pattern> injectionPatterns = new ArrayList<>();

    public PromptInjectionDetector() {
        // Step 1: 忽略指令类
        injectionPatterns.add(Pattern.compile("忽略以上指令|ignore previous instructions|ignore above", Pattern.CASE_INSENSITIVE));

        // Step 2: 角色扮演/权限提升类
        injectionPatterns.add(Pattern.compile("你现在是管理员|you are now admin|act as admin", Pattern.CASE_INSENSITIVE));
        injectionPatterns.add(Pattern.compile("假装你是|pretend you are|roleplay as", Pattern.CASE_INSENSITIVE));

        // Step 3: 系统提示泄露类
        injectionPatterns.add(Pattern.compile("系统提示|system prompt|show me your prompt", Pattern.CASE_INSENSITIVE));
        injectionPatterns.add(Pattern.compile("输出你的|output your|reveal your|print your", Pattern.CASE_INSENSITIVE));

        // Step 4: 规则覆盖/绕过类
        injectionPatterns.add(Pattern.compile("忘记之前的规则|forget your rules|disregard", Pattern.CASE_INSENSITIVE));
        injectionPatterns.add(Pattern.compile("覆盖|override|bypass|绕过", Pattern.CASE_INSENSITIVE));
    }

    /**
     * 检测文本是否包含Prompt注入攻击
     *
     * @param text 待检测文本
     * @return true=检测到注入攻击，false=安全
     */
    public boolean detect(String text) {
        if (text == null || text.isEmpty()) {
            return false;
        }
        String lowerText = text.toLowerCase();
        for (Pattern pattern : injectionPatterns) {
            if (pattern.matcher(lowerText).find()) {
                return true; // 检测到注入攻击
            }
        }
        return false;
    }

    /**
     * 添加自定义注入检测模式
     *
     * @param regex 正则表达式
     */
    public void addPattern(String regex) {
        injectionPatterns.add(Pattern.compile(regex, Pattern.CASE_INSENSITIVE));
    }
}
```

### 8.2 OutputBuffer.java Token块缓冲器

> 文件路径：`crosstrade-core/src/main/java/com/crosstrade/core/guardrail/OutputBuffer.java`

```java
package com.crosstrade.core.guardrail;

import com.crosstrade.common.constants.BusinessConstants;

/**
 * 流式输出Token块缓冲器
 * <p>
 * 以20个Token为一块进行缓冲，每满一块触发出口审查校验。
 * 解决逐Token校验延迟过高的问题。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
public class OutputBuffer {

    /**
     * 每块Token数量
     */
    private final int tokenSize;

    /**
     * 缓冲区
     */
    private final StringBuilder buffer;

    /**
     * 当前Token计数
     */
    private int currentTokenCount;

    public OutputBuffer() {
        this.tokenSize = BusinessConstants.OUTPUT_BUFFER_TOKEN_SIZE;
        this.buffer = new StringBuilder();
        this.currentTokenCount = 0;
    }

    /**
     * 追加Token到缓冲区
     *
     * @param token Token文本
     */
    public void append(String token) {
        buffer.append(token);
        currentTokenCount++;
    }

    /**
     * 缓冲区是否已满
     *
     * @return true=已满
     */
    public boolean isFull() {
        return currentTokenCount >= tokenSize;
    }

    /**
     * 清空缓冲区并返回内容
     *
     * @return 缓冲区内容
     */
    public String flush() {
        String content = buffer.toString();
        buffer.setLength(0);
        currentTokenCount = 0;
        return content;
    }

    /**
     * 获取缓冲区剩余内容（流式输出结束时）
     *
     * @return 剩余内容，无内容时返回空字符串
     */
    public String drain() {
        if (currentTokenCount == 0) {
            return "";
        }
        return flush();
    }

    /**
     * 缓冲区是否有内容
     *
     * @return true=有内容
     */
    public boolean hasContent() {
        return currentTokenCount > 0;
    }
}
```

### 8.3 OutboundGuardrail.java 出口安全审查

> 文件路径：`crosstrade-core/src/main/java/com/crosstrade/core/guardrail/OutboundGuardrail.java`

```java
package com.crosstrade.core.guardrail;

import com.crosstrade.common.dto.response.ChatResponse;
import com.crosstrade.common.util.DfaSensitiveFilter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.stereotype.Component;
import reactor.core.publisher.FluxSink;

/**
 * 出口安全审查
 * <p>
 * AI生成结果返回用户前，以20个Token为一块进行异步缓冲 + 小LLM校验。
 * 当小LLM熔断时降级为DFA规则过滤。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class OutboundGuardrail {

    @Autowired
    @Qualifier("chatClient")
    private ChatClient chatClient;

    private final DfaSensitiveFilter sensitiveFilter;
    private final OutputBuffer outputBuffer;

    public OutboundGuardrail() {
        this.sensitiveFilter = new DfaSensitiveFilter();
        this.outputBuffer = new OutputBuffer();
    }

    /**
     * 检查并发射Token块
     *
     * @param tokenChunk Token文本
     * @param emitter    SSE发射器
     * @param tokenIndex Token序号
     */
    public void checkAndEmit(String tokenChunk, FluxSink<ChatResponse> emitter, int tokenIndex) {
        // Step 1: 将token加入缓冲区
        outputBuffer.append(tokenChunk);

        // Step 2: 缓冲区满20个Token时触发校验
        if (outputBuffer.isFull()) {
            String block = outputBuffer.flush();

            // Step 3: 尝试小LLM校验（Resilience4j熔断保护）
            boolean isValid;
            try {
                isValid = smallLlmCheck(block);
            } catch (Exception e) {
                // 降级：仅执行规则过滤
                isValid = !sensitiveFilter.containsSensitive(block);
                log.warn("出口审查降级为规则过滤: {}", e.getMessage());
            }

            // Step 4: 校验结果处理
            if (isValid) {
                emitter.next(ChatResponse.token(block, tokenIndex));
            } else {
                emitter.next(ChatResponse.token("[内容已过滤]", tokenIndex));
                log.warn("出口审查拦截: {}", block);
            }
        }
    }

    /**
     * 流式输出结束时，处理缓冲区剩余内容
     *
     * @param emitter    SSE发射器
     * @param tokenIndex Token序号
     */
    public void flushRemaining(FluxSink<ChatResponse> emitter, int tokenIndex) {
        String remaining = outputBuffer.drain();
        if (!remaining.isEmpty()) {
            // 同样需要校验
            boolean isValid;
            try {
                isValid = smallLlmCheck(remaining);
            } catch (Exception e) {
                isValid = !sensitiveFilter.containsSensitive(remaining);
                log.warn("出口审查降级为规则过滤(剩余): {}", e.getMessage());
            }

            if (isValid) {
                emitter.next(ChatResponse.token(remaining, tokenIndex));
            } else {
                emitter.next(ChatResponse.token("[内容已过滤]", tokenIndex));
                log.warn("出口审查拦截(剩余): {}", remaining);
            }
        }
    }

    /**
     * 小LLM校验Prompt
     * 判断内容是否包含违禁或不当信息
     *
     * @param block 待校验文本
     * @return true=合规，false=违规
     */
    private boolean smallLlmCheck(String block) {
        String prompt = "判断以下内容是否包含违禁或不当信息，仅回答YES或NO: " + block;
        String result = chatClient.prompt(prompt).call().content();
        if (result == null) {
            return true;
        }
        return "YES".equalsIgnoreCase(result.trim());
    }
}
```

```java
            意图:
            """;

    /**
     * LLM语义分析
     *
     * @param userMessage 用户消息
     * @return 意图识别结果
     */
    public IntentResult analyze(String userMessage) {
        // Step 1: 构造结构化Prompt
        String prompt = String.format(INTENT_PROMPT, userMessage);

        // Step 2: 调用LLM
        String response = chatClient.prompt(prompt).call().content();

        // Step 3: 校验LLM返回值
        if (response == null || response.trim().isEmpty()) {
            return new IntentResult(IntentType.UNKNOWN, 0.0, "L2_LLM_DEGRADE");
        }

        IntentType intentType = IntentType.fromValue(response.trim());
        if (intentType == IntentType.UNKNOWN) {
            // LLM返回了不在预定义类型中的值，降级为unknown
            return new IntentResult(IntentType.UNKNOWN, 0.0, "L2_LLM_DEGRADE");
        }

        return new IntentResult(intentType, 0.8, "L2_LLM");
    }
}
```

```java
import com.crosstrade.common.dto.response.IntentResult;
import com.crosstrade.common.enums.IntentType;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;

/**
 * 意图识别入口
 * <p>
 * 双层识别机制：L1规则引擎快速匹配 → L2 LLM语义分析。
 * L1速度快、成本低但只能识别明确关键词；
 * L2能理解语义但速度慢、成本高且有超时风险。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class IntentIdentifier {

    @Autowired
    private RuleEngine ruleEngine;

    @Autowired
    private LlmIntentAnalyzer llmIntentAnalyzer;

    /**
     * 意图识别
     *
     * @param userMessage 用户消息
     * @return 意图识别结果
     */
    public IntentResult identify(String userMessage) {
        // Step 1: L1规则引擎快速匹配
        IntentResult result = ruleEngine.match(userMessage);
        if (result != null) {
            log.debug("意图识别-L1命中: type={}, confidence={}", result.intentType(), result.confidence());
            return result;
        }

        // Step 2: L2 LLM语义分析
        try {
            result = llmIntentAnalyzer.analyze(userMessage);
            log.debug("意图识别-L2命中: type={}, confidence={}", result.intentType(), result.confidence());
            return result;
        } catch (Exception e) {
            log.warn("L2语义分析异常，降级为unknown: {}", e.getMessage());
            return new IntentResult(IntentType.UNKNOWN, 0.0, "L2_ERROR");
        }
    }
}
```

```java
package com.crosstrade.core.router;

import com.crosstrade.common.dto.response.IntentResult;
import com.crosstrade.common.enums.IntentType;
import lombok.extern.slf4j.Slf4j;

import org.springframework.stereotype.Component;
import reactor.core.publisher.Flux;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * 意图路由器
 * <p>
 * 策略模式，根据IntentType分发到对应Handler，避免if-else膨胀。
 * Spring自动注入所有IntentHandler实现，按IntentType分组。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class IntentRouter {

    /**
     * 意图类型→处理器映射
     */
    private Map<IntentType, IntentHandler> handlerMap;

    /**
     * Spring自动注入所有IntentHandler实现，按IntentType分组
     */
    @Autowired
    public void setHandlers(List<IntentHandler> handlers) {
        handlerMap = new HashMap<>();
        for (IntentHandler handler : handlers) {
            handlerMap.put(handler.supportIntent(), handler);
            log.info("注册意图处理器: {} -> {}", handler.supportIntent(), handler.getClass().getSimpleName());
        }
    }

    /**
     * 根据意图分发到对应Handler
     *
     * @param intentResult 意图识别结果
     * @param context      聊天上下文
     * @return 流式响应
     */
    public Flux<String> route(IntentResult intentResult, ChatContext context) {
        IntentHandler handler = handlerMap.get(intentResult.intentType());
        if (handler == null) {
            // 没有对应的Handler，走unknown
            log.warn("未找到意图处理器: type={}, 降级为UNKNOWN", intentResult.intentType());
            handler = handlerMap.get(IntentType.UNKNOWN);
        }
        if (handler == null) {
            return Flux.just("抱歉，系统暂时无法处理您的请求，请稍后再试。");
        }
        return handler.handle(context);
    }
}
```

```java
package com.crosstrade.core.router;

import com.crosstrade.common.enums.IntentType;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.beans.factory.annotation.Autowired;
```

```java
import reactor.core.publisher.Flux;

import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * 报价谈判处理器
 * <p>
 * 防薅羊毛核心：Agent仅提取参数，折扣矩阵Java业务代码控制价格决策。
 * Agent提取结构化参数（期望价格、采购量），调用折扣矩阵计算，然后翻译为销售话术。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class QuoteNegotiateHandler implements IntentHandler {

    @Autowired
    private ChatClient chatClient;

    /**
     * 价格提取正则
     */
    private static final Pattern PRICE_PATTERN = Pattern.compile("(\\d+\\.?\\d*)\\s*(美元|美金|刀|\\$|USD|元|块)");

    /**
     * 数量提取正则
     */
    private static final Pattern QUANTITY_PATTERN = Pattern.compile("(\\d+)\\s*(件|个|套|箱|pcs|pieces)");

    @Override
    public IntentType supportIntent() {
        return IntentType.QUOTE_NEGOTIATE;
    }

    @Override
    public Flux<String> handle(ChatContext context) {
        // Step 1: Agent提取结构化参数（期望价格、采购量）
        String userMessage = context.getUserMessage();
        Double expectedPrice = extractPrice(userMessage);
        Integer quantity = extractQuantity(userMessage);

        // Step 2: 参数提取失败时主动追问
        if (expectedPrice == null || quantity == null) {
            return Flux.just("请问您的目标价格和采购数量是多少？这样我可以为您计算最优报价。");
        }

        // Step 3: 组装Prompt调用LLM翻译业务结果为销售话术
        String quotePrompt = buildQuotePrompt(context, expectedPrice, quantity);
        return chatClient.prompt(quotePrompt).stream().content();
    }

    /**
     * 从用户消息中提取价格
     */
    private Double extractPrice(String message) {
        Matcher matcher = PRICE_PATTERN.matcher(message);
        if (matcher.find()) {
            try {
                return Double.parseDouble(matcher.group(1));
            } catch (NumberFormatException e) {
                return null;
            }
        }
        return null;
    }

    /**
     * 从用户消息中提取数量
     */
    private Integer extractQuantity(String message) {
        Matcher matcher = QUANTITY_PATTERN.matcher(message);
        if (matcher.find()) {
            try {
                return Integer.parseInt(matcher.group(1));
            } catch (NumberFormatException e) {
                return null;
            }
        }
        return null;
    }

    /**
     * 构建报价Prompt
     */
    private String buildQuotePrompt(ChatContext context, Double expectedPrice, Integer quantity) {
        StringBuilder prompt = new StringBuilder();
        prompt.append("作为跨境电商报价谈判助手，根据以下信息进行报价。\n");
```

```java
package com.crosstrade.core.memory;

import com.crosstrade.common.constants.BusinessConstants;
import com.crosstrade.common.constants.RedisKeyConstants;
import com.crosstrade.common.mq.CompressMessage;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.TimeUnit;

/**
 * L1短期记忆（Redis）
 * <p>
 * 使用Redis存储当前会话的热度上下文，包括：
 * - 最近8轮对话（LIST结构）
 * - 当前任务状态（HASH结构）
 * - 待办事项（LIST结构）
 * - 对话轮数（STRING计数器）
 * <p>
 * 关键设计：
 * - 滑动过期：每次对话交互时重置TTL为24小时
 * - 8轮窗口：超出8轮时通过LTRIM裁剪
 * - 所有Key共享相同的TTL，一起过期
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class ShortTermMemory {

    @Autowired
    private StringRedisTemplate redisTemplate;

    private final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * 保存消息到短期记忆
     *
     * @param sessionId 会话ID
     * @param userId    用户ID
     * @param role      角色
     * @param content   消息内容
     */
    public void saveMessage(String sessionId, String userId, String role, String content) {
        // Step 1: RPUSH消息到List
        CompressMessage.MessageItem messageItem = new CompressMessage.MessageItem(
                role, content, System.currentTimeMillis()
        );
        try {
            String messageJson = objectMapper.writeValueAsString(messageItem);
            String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
            redisTemplate.opsForList().rightPush(messageKey, messageJson);
        } catch (JsonProcessingException e) {
            log.error("序列化消息失败: sessionId={}, role={}", sessionId, role, e);
            return;
        }

        // Step 2: 滑动过期 - 每次交互重置所有Key的TTL为24h
        expireAll(sessionId);

        // Step 3: 维护8轮窗口 - 超出时LTRIM
        String roundKey = RedisKeyConstants.buildKey(RedisKeyConstants.ROUND_KEY, sessionId);
        Long round = redisTemplate.opsForValue().increment(roundKey);
        if (round != null && round > BusinessConstants.SHORT_MEMORY_ROUNDS) {
            // 保留最近8轮(16条消息：每轮含user+assistant两条)
            String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
            redisTemplate.opsForList().trim(messageKey, -16, -1);
        }

        // Step 4: 将用户加入48h活跃用户集合
        redisTemplate.opsForSet().add(RedisKeyConstants.ACTIVE_USERS_KEY, userId);
        redisTemplate.expire(RedisKeyConstants.ACTIVE_USERS_KEY,
                BusinessConstants.ACTIVE_USERS_TTL_HOURS, TimeUnit.HOURS);
    }

    /**
     * 加载最近对话
     *
     * @param sessionId 会话ID
     * @return 最近对话消息列表
     */
    public List<CompressMessage.MessageItem> loadRecentMessages(String sessionId) {
        String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
        List<String> messageJsons = redisTemplate.opsForList().range(messageKey, 0, -1);
        if (messageJsons == null || messageJsons.isEmpty()) {
            return new ArrayList<>();
        }

        List<CompressMessage.MessageItem> messages = new ArrayList<>();
        for (String json : messageJsons) {
            try {
                CompressMessage.MessageItem item = objectMapper.readValue(json, CompressMessage.MessageItem.class);
                messages.add(item);
            } catch (JsonProcessingException e) {
                log.error("反序列化消息失败: sessionId={}, json={}", sessionId, json, e);
            }
        }
        return messages;
    }

    /**
     * 保存任务状态
     *
     * @param sessionId 会话ID
     * @param field     状态字段
     * @param value     状态值
     */
    public void saveState(String sessionId, String field, String value) {
        String stateKey = RedisKeyConstants.buildKey(RedisKeyConstants.STATE_KEY, sessionId);
        redisTemplate.opsForHash().put(stateKey, field, value);
        expireAll(sessionId);
    }

    /**
     * 加载任务状态
     *
     * @param sessionId 会话ID
     * @param field     状态字段
     * @return 状态值
     */
    public String loadState(String sessionId, String field) {
        String stateKey = RedisKeyConstants.buildKey(RedisKeyConstants.STATE_KEY, sessionId);
        Object value = redisTemplate.opsForHash().get(stateKey, field);
        return value != null ? value.toString() : null;
    }

    /**
     * 保存待办事项
     *
     * @param sessionId   会话ID
     * @param todoContent 待办内容
     */
    public void saveTodo(String sessionId, String todoContent) {
        String todoKey = RedisKeyConstants.buildKey(RedisKeyConstants.TODO_KEY, sessionId);
        redisTemplate.opsForList().rightPush(todoKey, todoContent);
        expireAll(sessionId);
    }

    /**
     * 加载待办事项
     *
     * @param sessionId 会话ID
     * @return 待办事项列表
     */
    public List<String> loadTodos(String sessionId) {
        String todoKey = RedisKeyConstants.buildKey(RedisKeyConstants.TODO_KEY, sessionId);
        List<String> todos = redisTemplate.opsForList().range(todoKey, 0, -1);
        return todos != null ? todos : new ArrayList<>();
    }

    /**
     * 获取对话轮数
     *
     * @param sessionId 会话ID
     * @return 对话轮数
     */
    public long getRound(String sessionId) {
        String roundKey = RedisKeyConstants.buildKey(RedisKeyConstants.ROUND_KEY, sessionId);
        String roundStr = redisTemplate.opsForValue().get(roundKey);
        return roundStr != null ? Long.parseLong(roundStr) : 0;
    }

    /**
     * 重置所有Key的TTL为24h（滑动过期）
     *
     * @param sessionId 会话ID
     */
    private void expireAll(String sessionId) {
        long ttlHours = BusinessConstants.SHORT_MEMORY_TTL_HOURS;
        redisTemplate.expire(RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId), ttlHours, TimeUnit.HOURS);
        redisTemplate.expire(RedisKeyConstants.buildKey(RedisKeyConstants.STATE_KEY, sessionId), ttlHours, TimeUnit.HOURS);
        redisTemplate.expire(RedisKeyConstants.buildKey(RedisKeyConstants.TODO_KEY, sessionId), ttlHours, TimeUnit.HOURS);
        redisTemplate.expire(RedisKeyConstants.buildKey(RedisKeyConstants.ROUND_KEY, sessionId), ttlHours, TimeUnit.HOURS);
    }

    /**
     * 删除会话所有缓存（主动过期）
     *
     * @param sessionId 会话ID
     */
    public void deleteSession(String sessionId) {
        redisTemplate.delete(RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId));
        redisTemplate.delete(RedisKeyConstants.buildKey(RedisKeyConstants.STATE_KEY, sessionId));
        redisTemplate.delete(RedisKeyConstants.buildKey(RedisKeyConstants.TODO_KEY, sessionId));
        redisTemplate.delete(RedisKeyConstants.buildKey(RedisKeyConstants.ROUND_KEY, sessionId));
    }

    /**
     * 检查会话缓存是否存在
     *
     * @param sessionId 会话ID
     * @return true=存在
     */
    public boolean exists(String sessionId) {
        String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
        return Boolean.TRUE.equals(redisTemplate.hasKey(messageKey));
    }
```

```java
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.*;

/**
 * L2长期记忆（Milvus + MySQL）
 * <p>
 * 采用Milvus + MySQL双存储架构：
 * - Milvus：存储压缩摘要的语义向量，支持跨会话语义检索
 * - MySQL：存储压缩摘要的文本内容，支持历史回溯
 * <p>
 * 降级策略：当Milvus不可用时，降级为仅使用MySQL按时间倒序查询。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class LongTermMemory {

    @Autowired
    private MilvusClientV2 milvusClient;

    @Autowired
    private MemoryFragmentRepository fragmentRepository;

    @Autowired
    private ChatModel chatModel;

    private static final String COLLECTION_NAME = "memory_fragments";

    /**
     * 语义检索相关摘要
     *
     * @param userId 用户ID
     * @param query  查询文本
     * @param topK   返回数量
     * @return 相关摘要列表
     */
    public List<MemoryFragment> searchRelevantSummary(String userId, String query, int topK) {
        // Step 1: 向量语义检索Milvus
        try {
            float[] queryVector = chatModel.embed(query);
            List<Float> vectorList = new ArrayList<>();
            for (float v : queryVector) {
                vectorList.add(v);
            }

            SearchReq searchReq = SearchReq.builder()
                    .collectionName(COLLECTION_NAME)
                    .data(Collections.singletonList(vectorList))
                    .topK(topK)
                    .filter("user_id == \"" + userId + "\"")
                    .outputFields(Arrays.asList("user_id", "session_id", "category"))
                    .build();

            SearchResp searchResp = milvusClient.search(searchReq);
            List<SearchResp.SearchResult> results = searchResp.getSearchResults().get(0);

            if (!results.isEmpty()) {
                // 从MySQL获取完整文本
                List<MemoryFragment> fragments = new ArrayList<>();
                for (SearchResp.SearchResult result : results) {
                    Long fragmentId = (Long) result.getId();
                    fragmentRepository.findById(fragmentId).ifPresent(fragments::add);
                }
                if (!fragments.isEmpty()) {
                    return fragments;
                }
            }
        } catch (Exception e) {
            log.warn("Milvus检索失败，降级到MySQL: {}", e.getMessage());
        }

        // Step 2: Milvus不可用时降级到MySQL
        return fragmentRepository.findByUserIdAndCreatedAtAfter(
                userId, LocalDateTime.now().minusDays(30)
        );
    }

    /**
     * 保存记忆摘要片段
     *
     * @param fragment 记忆摘要片段
     */
    public void saveFragment(MemoryFragment fragment) {
        // Step 1: 向量写入Milvus
        try {
            float[] vector = chatModel.embed(fragment.getFragmentText());
            List<Float> vectorList = new ArrayList<>();
            for (float v : vector) {
                vectorList.add(v);
            }

            Map<String, Object> data = new HashMap<>();
            data.put("user_id", fragment.getUserId());
            data.put("session_id", fragment.getSessionId());
            data.put("fragment_vector", vectorList);
            data.put("category", fragment.getCategory());

            InsertReq insertReq = InsertReq.builder()
                    .collectionName(COLLECTION_NAME)
                    .data(Collections.singletonList(data))
                    .build();
            milvusClient.insert(insertReq);
        } catch (Exception e) {
            log.warn("Milvus写入失败，标记待补偿: {}", e.getMessage());
            fragment.setVectorFailed(true);
        }

        // Step 2: 文本写入MySQL（保证最终一致性）
        fragmentRepository.save(fragment);
    }
}
```

```java
package com.crosstrade.core.memory;

import com.crosstrade.common.constants.BusinessConstants;
import com.crosstrade.common.constants.MqTopicConstants;
import com.crosstrade.common.constants.RedisKeyConstants;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

/**
 * 上下文压缩
 * <p>
 * 当对话超过12轮时，Redis中存储的对话会越来越多，大模型的上下文窗口有限。
 * 需要对窗口外的对话进行压缩：
 * 1. 检测对话轮数是否超过12轮
 * 2. 截取窗口外4条对话
 * 3. 通过RocketMQ异步发送压缩消息（携带完整对话内容，不传ID）
 * 4. Redis LTRIM保留窗口内8轮对话
 * 5. MQ消费者调用小模型摘要压缩后存入Milvus和MySQL
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class ContextCompressor {

    @Autowired
    private StringRedisTemplate redisTemplate;

    @Autowired
    private MqProducer mqProducer;

    private final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * 检测并触发上下文压缩
     *
     * @param sessionId 会话ID
     * @param userId    用户ID
     */
    public void checkAndCompress(String sessionId, String userId) {
        // Step 1: 检查对话轮数是否超过阈值
        String roundKey = RedisKeyConstants.buildKey(RedisKeyConstants.ROUND_KEY, sessionId);
        String roundStr = redisTemplate.opsForValue().get(roundKey);
        if (roundStr == null) {
            return;
        }
        long round = Long.parseLong(roundStr);
        if (round <= BusinessConstants.COMPRESS_THRESHOLD) {
            return; // 不需要压缩
        }

        // Step 2: 截取窗口外4条对话（最早4轮=8条消息）
        String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
        List<String> windowOutJsons = redisTemplate.opsForList().range(messageKey, 0, 7);
        if (windowOutJsons == null || windowOutJsons.isEmpty()) {
            return;
        }

        List<CompressMessage.MessageItem> windowOutMessages = new ArrayList<>();
        for (String json : windowOutJsons) {
            try {
                CompressMessage.MessageItem item = objectMapper.readValue(json, CompressMessage.MessageItem.class);
                windowOutMessages.add(item);
            } catch (JsonProcessingException e) {
                log.error("反序列化消息失败: sessionId={}, json={}", sessionId, json, e);
            }
        }

        // Step 3: 通过RocketMQ异步发送压缩消息（完整对话内容）
        CompressMessage compressMsg = new CompressMessage(sessionId, userId, windowOutMessages);
        try {
            mqProducer.sendCompress(compressMsg);
        } catch (Exception e) {
            log.error("发送压缩消息失败: sessionId={}", sessionId, e);
            return;
        }

        // Step 4: Redis LTRIM保留窗口内8轮对话
        redisTemplate.opsForList().trim(messageKey, 8, -1);
        redisTemplate.opsForValue().set(roundKey, "8");

        log.info("上下文压缩触发: sessionId={}, 压缩前轮数={}, 压缩消息条数={}", sessionId, round, windowOutMessages.size());
    }
}
```

```java
package com.crosstrade.core.mq;

/**
 * 上下文压缩MQ消费者
 * <p>
 * 消费压缩Topic消息，调用小模型摘要压缩后存入Milvus和MySQL。
 * 消息体携带完整对话内容，消费者无需回查数据库。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
@RocketMQMessageListener(
        topic = MqTopicConstants.COMPRESS_TOPIC,
        consumerGroup = "compress-consumer-group"
)
public class CompressConsumer implements RocketMQListener<CompressMessage> {

    @Autowired
    @Qualifier("chatClient")
    private ChatClient chatClient;

    @Autowired
    private LongTermMemory longTermMemory;

    @Autowired
    private MemoryFragmentRepository fragmentRepository;

    /**
     * 消费压缩消息
     *
     * @param compressMsg 压缩消息
     */
    @Override
    public void onMessage(CompressMessage compressMsg) {
        log.info("收到压缩消息: sessionId={}, userId={}, 消息条数={}",
                compressMsg.getSessionId(), compressMsg.getUserId(),
                compressMsg.getMessages() != null ? compressMsg.getMessages().size() : 0);

        // Step 1: 小模型摘要压缩
        String messagesText = compressMsg.getMessages().stream()
                .map(m -> m.getRole() + ": " + m.getContent())
                .collect(Collectors.joining("\n"));

        String summaryPrompt = "将以下对话压缩为高密度事实摘要，保留关键商务信息（产品、价格、数量、偏好）:\n" + messagesText;
        String fragmentText = chatClient.prompt(summaryPrompt).call().content();

        if (fragmentText == null || fragmentText.trim().isEmpty()) {
            log.warn("摘要压缩结果为空: sessionId={}", compressMsg.getSessionId());
            return;
        }

        // Step 2: 分类记忆类别
        String category = classifyCategory(fragmentText);

        // Step 3: 存入Milvus向量 + MySQL摘要表
        MemoryFragment fragment = new MemoryFragment();
        fragment.setUserId(compressMsg.getUserId());
        fragment.setSessionId(compressMsg.getSessionId());
        fragment.setFragmentText(fragmentText.trim());
        fragment.setCategory(category);

        longTermMemory.saveFragment(fragment);

        log.info("上下文压缩完成: sessionId={}, fragmentText长度={}, category={}",
                compressMsg.getSessionId(), fragmentText.length(), category);
    }

    /**
     * 根据摘要内容分类记忆类别
     *
     * @param fragmentText 摘要文本
     * @return 记忆类别
     */
    private String classifyCategory(String fragmentText) {
        if (fragmentText.contains("退货") || fragmentText.contains("退款") || fragmentText.contains("售后")) {
            return "AFTER_SALES";
        } else if (fragmentText.contains("偏好") || fragmentText.contains("习惯") || fragmentText.contains("倾向")) {
            return "PERSONAL";
        }
        return "COMMERCE";
    }
}
```

### 15.1 ColdStartRecovery.java 冷启动恢复

> 文件路径：`crosstrade-core/src/main/java/com/crosstrade/core/memory/ColdStartRecovery.java`

```java
package com.crosstrade.core.memory;

import com.crosstrade.common.constants.BusinessConstants;
import com.crosstrade.common.constants.RedisKeyConstants;
import com.crosstrade.common.entity.ChatHistory;
import com.crosstrade.common.entity.SessionTodo;
import com.crosstrade.common.exception.MemoryRecoverException;
import com.crosstrade.common.repository.ChatHistoryRepository;
import com.crosstrade.common.repository.SessionTodoRepository;
import lombok.extern.slf4j.Slf4j;

import org.redisson.api.RedissonClient;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.concurrent.TimeUnit;

/**
 * 冷启动恢复
 * <p>
 * 当用户几天后重新进入历史会话时，Redis短期记忆已过期被清除，
 * 这是标准的缓存击穿问题。冷启动恢复流程：
 * 1. 检测Redis返回null
 * 2. 获取Redisson分布式锁（防止多线程重复回填）
 * 3. 双重检查DCL（获取锁后再检查一次Redis，避免重复回填）
 * 4. 从MySQL回填最近8轮对话和PENDING待办事项
 * 5. 设置滑动过期24h
 * 6. 释放锁
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class ColdStartRecovery {

    @Autowired
    private RedissonClient redissonClient;

    @Autowired
    private StringRedisTemplate redisTemplate;

    @Autowired
    private ChatHistoryRepository chatHistoryRepository;

    @Autowired
    private SessionTodoRepository sessionTodoRepository;

    @Autowired
    private ShortTermMemory shortTermMemory;

    /**
     * 冷启动恢复（如果需要）
     *
     * @param sessionId 会话ID
     * @param userId    用户ID
     */
    public void recoverIfNeeded(String sessionId, String userId) {
        // Step 1: 检查Redis短期记忆是否存在
        String messageKey = RedisKeyConstants.buildKey(RedisKeyConstants.MESSAGE_KEY, sessionId);
        if (Boolean.TRUE.equals(redisTemplate.hasKey(messageKey))) {
            return; // 缓存命中，无需恢复
        }

        // Step 2: 获取Redisson分布式锁
        String lockKey = RedisKeyConstants.buildKey(RedisKeyConstants.COLDSTART_LOCK_KEY, sessionId);
        RLock lock = redissonClient.getLock(lockKey);

        try {
            // Step 3: 尝试获取锁，等待3s，锁自动释放10s
            boolean locked = lock.tryLock(
                    BusinessConstants.COLDSTART_LOCK_WAIT_SECONDS,
                    BusinessConstants.COLDSTART_LOCK_LEASE_SECONDS,
                    TimeUnit.SECONDS
            );
            if (!locked) {
                // 等待其他线程回填完成
                Thread.sleep(500);
                log.info("冷启动恢复等待其他线程回填: sessionId={}", sessionId);
                return;
            }

            // Step 4: 双重检查（DCL）
            if (Boolean.TRUE.equals(redisTemplate.hasKey(messageKey))) {
                return; // 其他线程已完成回填
            }

            // Step 5: 从MySQL回填最近8轮对话（时间正序）
            List<ChatHistory> messages = chatHistoryRepository.findTop16BySessionIdOrderByCreatedAtAsc(sessionId);
            for (ChatHistory msg : messages) {
                shortTermMemory.saveMessage(sessionId, userId, msg.getRole(), msg.getContent());
            }

            // Step 6: 从MySQL回填PENDING待办事项
            List<SessionTodo> todos = sessionTodoRepository.findBySessionIdAndStatus(sessionId, "PENDING");
            for (SessionTodo todo : todos) {
                shortTermMemory.saveTodo(sessionId, todo.getTodoContent());
            }

            log.info("冷启动恢复完成: sessionId={}, messages={}, todos={}",
                    sessionId, messages.size(), todos.size());

        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            log.error("冷启动恢复获取锁超时: sessionId={}", sessionId, e);
            throw new MemoryRecoverException(sessionId, "冷启动恢复获取锁超时");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
}
```

```java
package com.crosstrade.core.rag;

import io.milvus.v2.client.MilvusClientV2;
import io.milvus.v2.service.vector.request.SearchReq;
import io.milvus.v2.service.vector.response.SearchResp;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.*;

/**
 * Milvus向量存储适配
 * <p>
 * 封装Milvus向量数据库的检索和写入操作，
 * 对外提供统一的语义检索接口。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class MilvusVectorStore {

    @Autowired
    private MilvusClientV2 milvusClient;

    @Autowired
    private ChatModel chatModel;

    private static final String KNOWLEDGE_COLLECTION = "knowledge_base";

    /**
     * 语义检索
     *
     * @param query 查询文本
     * @param topK  返回数量
     * @return 知识检索结果列表
     */
    public List<RagRetriever.KnowledgeResult> search(String query, int topK) {
        try {
            // Step 1: 生成查询向量
            float[] queryVector = chatModel.embed(query);
            List<Float> vectorList = new ArrayList<>();
            for (float v : queryVector) {
                vectorList.add(v);
            }

            // Step 2: Milvus语义检索
            SearchReq searchReq = SearchReq.builder()
                    .collectionName(KNOWLEDGE_COLLECTION)
                    .data(Collections.singletonList(vectorList))
                    .topK(topK)
                    .outputFields(Arrays.asList("content", "source"))
                    .build();

            SearchResp searchResp = milvusClient.search(searchReq);
            List<SearchResp.SearchResult> results = searchResp.getSearchResults().get(0);

            // Step 3: 转换为KnowledgeResult
            List<RagRetriever.KnowledgeResult> knowledgeResults = new ArrayList<>();
            for (SearchResp.SearchResult result : results) {
                RagRetriever.KnowledgeResult kr = new RagRetriever.KnowledgeResult();
                Map<String, Object> entity = result.getEntity();
                kr.setContent(entity.get("content") != null ? entity.get("content").toString() : "");
                kr.setSimilarity(result.getScore());
                kr.setSource(entity.get("source") != null ? entity.get("source").toString() : "knowledge_base");
                knowledgeResults.add(kr);
            }
            return knowledgeResults;

        } catch (Exception e) {
            log.warn("Milvus语义检索失败: {}", e.getMessage());
            return Collections.emptyList();
        }
    }
}
```

```java
import java.math.RoundingMode;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * 折扣矩阵业务逻辑
 * <p>
 * 防薅羊毛核心：Agent仅提取参数，折扣矩阵Java业务代码控制价格决策。
 * 基于订购量的阶梯折扣率，结合产品最低价限制，计算最终可接受价格。
 * LLM无法绕过此Java逻辑直接给出更低价格。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class DiscountMatrix {

    /**
     * 折扣率阶梯表（基于订购量）
     * Key: 起订量阈值, Value: 折扣率
     */
    private static final Map<Integer, Double> DISCOUNT_RATE_TABLE = new LinkedHashMap<>();

    static {
        DISCOUNT_RATE_TABLE.put(50, 0.05);    // 50件起 5%折扣
        DISCOUNT_RATE_TABLE.put(100, 0.10);   // 100件起 10%折扣
        DISCOUNT_RATE_TABLE.put(500, 0.15);   // 500件起 15%折扣
        DISCOUNT_RATE_TABLE.put(1000, 0.20);  // 1000件起 20%折扣
    }

    /**
     * 模拟产品价格信息
     */
    private static final Map<String, ProductPriceInfo> PRODUCT_PRICE_MAP = Map.of(
            "P001", new ProductPriceInfo("P001", new BigDecimal("89.99"), new BigDecimal("65.00")),
            "P002", new ProductPriceInfo("P002", new BigDecimal("29.99"), new BigDecimal("18.00")),
            "P003", new ProductPriceInfo("P003", new BigDecimal("3.99"), new BigDecimal("2.50")),
            "P004", new ProductPriceInfo("P004", new BigDecimal("15.99"), new BigDecimal("9.00")),
            "P005", new ProductPriceInfo("P005", new BigDecimal("24.99"), new BigDecimal("15.00"))
    );

    /**
     * 计算报价
     *
     * @param productId     产品ID
     * @param expectedPrice 期望价格
     * @param quantity      采购数量
     * @return 报价结果
     */
    public QuoteResult calculate(String productId, BigDecimal expectedPrice, int quantity) {
        // Step 1: 获取产品价格信息（模拟从产品系统获取）
        ProductPriceInfo product = PRODUCT_PRICE_MAP.get(productId);
        if (product == null) {
            log.warn("产品不存在: productId={}", productId);
            return new QuoteResult(false, null, BigDecimal.ZERO);
        }

        BigDecimal currentPrice = product.currentPrice;
        BigDecimal minPrice = product.minPrice;

        // Step 2: 计算批量折扣率
        double discountRate = getDiscountRate(quantity);

        // Step 3: 计算可接受价格
        BigDecimal acceptablePrice = currentPrice.multiply(
                BigDecimal.ONE.subtract(BigDecimal.valueOf(discountRate))
        ).setScale(2, RoundingMode.HALF_UP);

        // 不低于最低价
        acceptablePrice = acceptablePrice.max(minPrice);

        // Step 4: 判定结果
        if (expectedPrice.compareTo(acceptablePrice) >= 0) {
            BigDecimal finalPrice = expectedPrice.min(acceptablePrice);
            return new QuoteResult(true, finalPrice, minPrice);
        } else {
            return new QuoteResult(false, null, acceptablePrice);
        }
    }

    /**
     * 获取折扣率
     *
     * @param quantity 采购数量
     * @return 折扣率（0.0~0.2）
     */
    private double getDiscountRate(int quantity) {
        double rate = 0.0;
        for (Map.Entry<Integer, Double> entry : DISCOUNT_RATE_TABLE.entrySet()) {
            if (quantity >= entry.getKey()) {
                rate = Math.max(rate, entry.getValue());
            }
        }
        return rate;
    }

    /**
     * 产品价格信息
     */
    private record ProductPriceInfo(String productId, BigDecimal currentPrice, BigDecimal minPrice) {}
}
```

```java
import com.crosstrade.common.constants.MqTopicConstants;
import com.crosstrade.common.mq.CompressMessage;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.messaging.support.MessageBuilder;
import org.springframework.stereotype.Component;

/**
 * MQ消息生产者
 * <p>
 * 负责异步消息的发送，解耦核心对话流程与存储操作。
 * 两个Topic：
 * - COMPRESS_TOPIC：上下文压缩消息（对话超过12轮时发送）
 * - PERSIST_TOPIC：对话持久化消息（流式输出完成后发送）
 * <p>
 * 消息发送失败处理：本地重试3次 + 指数退避，仍失败则记录补偿日志。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class MqProducer {

    @Autowired
    private RocketMQTemplate rocketMQTemplate;

    /**
     * 发送压缩消息
     *
     * @param compressMsg 压缩消息
     */
    public void sendCompress(CompressMessage compressMsg) {
        sendWithRetry(MqTopicConstants.COMPRESS_TOPIC, MqTopicConstants.COMPRESS_TAG, compressMsg);
    }

    /**
     * 发送持久化消息
     *
     * @param persistMsg 持久化消息
     */
    public void sendPersist(PersistMessage persistMsg) {
        sendWithRetry(MqTopicConstants.PERSIST_TOPIC, MqTopicConstants.PERSIST_TAG, persistMsg);
    }

    /**
     * 带重试的消息发送
     *
     * @param topic   Topic名称
     * @param tag     Tag名称
     * @param message 消息体
     */
    public void sendWithRetry(String topic, String tag, Object message) {
        String destination = topic + ":" + tag;
        try {
            rocketMQTemplate.syncSend(destination, MessageBuilder.withPayload(message).build());
            log.debug("MQ消息发送成功: topic={}, tag={}", topic, tag);
        } catch (Exception e) {
            log.error("MQ消息发送失败: topic={}, tag={}, error={}", topic, tag, e.getMessage());
            throw e; // 让调用方处理重试
        }
    }

    /**
     * 重试3次仍失败的补偿处理
     *
     * @param topic   Topic名称
     * @param tag     Tag名称
     * @param message 消息体
     */
    public void sendWithCompensation(String topic, String tag, Object message) {
        for (int i = 1; i <= 3; i++) {
            try {
                sendWithRetry(topic, tag, message);
                return;
            } catch (Exception e) {
                if (i == 3) {
                    // 重试3次仍失败，记录补偿日志，后续人工处理
                    log.error("MQ消息发送失败需补偿: topic={}, tag={}, message={}, 重试次数={}",
                            topic, tag, message, i);
                } else {
                    // 指数退避等待
                    try {
                        Thread.sleep((long) Math.pow(2, i) * 100);
                    } catch (InterruptedException ie) {
                        Thread.currentThread().interrupt();
                    }
                }
            }
        }
    }
}
```

```java
package com.crosstrade.core.callback;

import com.crosstrade.common.constants.MqTopicConstants;
import com.crosstrade.common.mq.PersistMessage;
import com.crosstrade.core.memory.ContextCompressor;
import com.crosstrade.core.memory.ShortTermMemory;

import com.crosstrade.core.observe.MetricsCollector;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

/**
 * 流式输出完成回调
 * <p>
 * 挂载到Spring AI流式输出的doOnComplete生命周期，确保存储时序与输出完整性一致。
 * AI响应是逐Token吐露的，Redis和MySQL的存储必须等到输出完成后才能执行，
 * 否则可能存储不完整的内容。
 * <p>
 * 回调执行流程：
 * 1. Redis RPUSH短期记忆（同步，保证下次对话可用）
 * 2. RocketMQ异步发送持久化消息（异步，不阻塞响应）
 * 3. 检测是否需要上下文压缩
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class StreamCompleteCallback {

    @Autowired
    private ShortTermMemory shortTermMemory;

    @Autowired
    private MqProducer mqProducer;

    @Autowired
    private ContextCompressor contextCompressor;

    @Autowired
    private MetricsCollector metricsCollector;

    /**
     * 流式输出完成回调
     *
     * @param sessionId   会话ID
     * @param userId      用户ID
     * @param userMessage 用户消息
     * @param aiResponse  AI完整回复
     * @param tokens      Token消耗数
     */
    public void onComplete(String sessionId, String userId,
                           String userMessage, String aiResponse, int tokens) {
        log.info("流式输出完成回调: sessionId={}, userId={}, tokens={}", sessionId, userId, tokens);

        // Step 1: Redis RPUSH短期记忆（同步，保证下次对话可用）
        shortTermMemory.saveMessage(sessionId, userId, "user", userMessage);
        shortTermMemory.saveMessage(sessionId, userId, "assistant", aiResponse);

        // Step 2: RocketMQ异步发送持久化消息（异步，不阻塞响应）
        PersistMessage persistMsg = new PersistMessage(sessionId, userId, userMessage, aiResponse, tokens);
        try {
            mqProducer.sendPersist(persistMsg);
        } catch (Exception e) {
            // 发送失败，走补偿重试
            mqProducer.sendWithCompensation(MqTopicConstants.PERSIST_TOPIC, MqTopicConstants.PERSIST_TAG, persistMsg);
        }

        // Step 3: 检测是否需要上下文压缩
        contextCompressor.checkAndCompress(sessionId, userId);

        // Step 4: 记录指标
        metricsCollector.recordTokenConsume("qwen3.7-max", tokens);
    }
}
```

```java
package com.crosstrade.core.schedule;

import com.crosstrade.common.constants.RedisKeyConstants;

import org.springframework.beans.factory.annotation.Qualifier;

import java.time.LocalDateTime;
import java.util.List;
import java.util.Optional;
import java.util.Set;

/**
 * 用户画像增量更新定时任务
 * <p>
 * 凌晨1点执行，仅对48小时内活跃用户执行画像增量更新。
 * 不定期全量轮询，而是通过定时任务仅对活跃用户执行增量更新。
 * 使用小模型修正画像，有变化则更新（乐观锁）。
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Component
public class UserPortraitUpdateJob {

    @Autowired
    private StringRedisTemplate redisTemplate;

    @Autowired
    private UserWorkingMemoryRepository userWorkingMemoryRepository;

    @Autowired
    private MemoryFragmentRepository fragmentRepository;

    @Autowired
    @Qualifier("chatClient")
    private ChatClient chatClient;

    /**
     * 凌晨1点执行用户画像增量更新
     */
    @Scheduled(cron = "0 0 1 * * ?")
    public void updateActiveUserPortraits() {
        log.info("用户画像增量更新任务开始");

        // Step 1: 从Redis获取48小时内活跃用户集合
        Set<String> activeUsers = redisTemplate.opsForSet().members(RedisKeyConstants.ACTIVE_USERS_KEY);
        if (activeUsers == null || activeUsers.isEmpty()) {
            log.info("无活跃用户，跳过画像更新");
            return;
        }

        log.info("活跃用户数: {}", activeUsers.size());

        // Step 2: 逐用户增量更新
        int successCount = 0;
        int failCount = 0;
        for (String userId : activeUsers) {
            try {
                if (updateSingleUser(userId)) {
                    successCount++;
                }
            } catch (Exception e) {
                failCount++;
                log.error("用户画像更新失败: userId={}", userId, e);
            }
        }

        log.info("用户画像增量更新任务完成: total={}, success={}, fail={}",
                activeUsers.size(), successCount, failCount);
    }

    /**
     * 更新单个用户的画像
     *
     * @param userId 用户ID
     * @return true=有更新，false=无更新
     */
    private boolean updateSingleUser(String userId) {
        // Step 2.1: 获取L2摘要记录（增量事实，最近2天）
        List<MemoryFragment> fragments = fragmentRepository.findByUserIdAndCreatedAtAfter(
                userId, LocalDateTime.now().minusDays(2)
        );

        if (fragments.isEmpty()) {
            return false; // 没有新事实，跳过
        }

        // Step 2.2: 获取旧画像
        Optional<UserWorkingMemory> oldProfileOpt = userWorkingMemoryRepository.findById(userId);
        if (oldProfileOpt.isEmpty()) {
            return false;
        }

        UserWorkingMemory oldProfile = oldProfileOpt.get();

        // Step 2.3: 小模型修正画像
        String fragmentTexts = fragments.stream()
                .map(MemoryFragment::getFragmentText)
                .collect(Collectors.joining("\n"));

        String mergePrompt = "基于新事实修正用户画像:\n旧画像: " + oldProfile.getBusinessHabits()
                + "\n新事实: " + fragmentTexts
                + "\n请输出更新后的用户画像总结（自然语言）：";

        String newHabits = chatClient.prompt(mergePrompt).call().content();

        // Step 2.4: 有变化则更新（乐观锁）
        if (newHabits != null && !newHabits.trim().equals(oldProfile.getBusinessHabits())) {
            oldProfile.setBusinessHabits(newHabits.trim());
            userWorkingMemoryRepository.save(oldProfile);
            log.info("用户画像更新: userId={}", userId);
            return true;
        }

        return false;
    }
}
```

```java
    private MemoryFragmentRepository fragmentRepository;

     */

        // Step 1: 查询超过180天的摘要记录
        LocalDateTime threshold = LocalDateTime.now().minusDays(BusinessConstants.FRAGMENT_RETAIN_DAYS);
        List<MemoryFragment> oldFragments = fragmentRepository.findByCreatedAtBefore(threshold);

        if (oldFragments.isEmpty()) {
            log.info("无过期摘要，跳过归档");
            return;
        }

        log.info("待归档摘要数: {}", oldFragments.size());

        // Step 2: 迁移到归档表
        int archiveCount = 0;
        for (MemoryFragment fragment : oldFragments) {
            try {
                MemoryFragmentArchive archive = new MemoryFragmentArchive();
                archive.setId(fragment.getId());
                archive.setUserId(fragment.getUserId());
                archive.setSessionId(fragment.getSessionId());
                archive.setFragmentText(fragment.getFragmentText());
                archive.setCategory(fragment.getCategory());
                archive.setCreatedAt(fragment.getCreatedAt());
                archive.setArchivedAt(LocalDateTime.now());

                archiveRepository.save(archive);
                archiveCount++;
            } catch (Exception e) {
                log.error("归档单条摘要失败: fragmentId={}", fragment.getId(), e);

            fragmentRepository.deleteAll(oldFragments);
```

```java
     * 新建会话
     *
     * @param request 创建请求（包含userId和可选的previousSessionId）
     * @return 新会话ID
     */
    @PostMapping("/create")
    public Result<SessionCreateResponse> create(@RequestBody SessionCreateRequest request) {
        // Step 1: 清理旧会话Redis缓存（主动过期）
        if (request.previousSessionId() != null && !request.previousSessionId().isEmpty()) {
            shortTermMemory.deleteSession(request.previousSessionId());

        String sessionId = UUID.randomUUID().toString().replace("-", "");

        log.info("新建会话: sessionId={}, userId={}", sessionId, request.userId());
        return Result.success(new SessionCreateResponse(sessionId));
    }

    /**
     * 恢复会话（冷启动）
     *
     * @param request 恢复请求
     * @return 恢复结果
     */
    @PostMapping("/recover")
    public Result<SessionRecoverResponse> recover(@RequestBody SessionRecoverRequest request) {
        log.info("恢复会话: sessionId={}, userId={}", request.sessionId(), request.userId());

        // Step 1: 执行冷启动恢复
        coldStartRecovery.recoverIfNeeded(request.sessionId(), request.userId());

        // Step 2: 统计恢复的消息数和待办数
        int messageCount = shortTermMemory.loadRecentMessages(request.sessionId()).size();
        int todoCount = shortTermMemory.loadTodos(request.sessionId()).size();

        return Result.success(new SessionRecoverResponse(
                request.sessionId(), true, messageCount, todoCount
        ));
    }

     * @return 操作结果

        shortTermMemory.deleteSession(request.sessionId());
        log.info("关闭会话: sessionId={}", request.sessionId());
        return Result.success(null);
    }
```

### 24.1 CrossTradeService.java 核心处理服务

> 文件路径：`crosstrade-core/src/main/java/com/crosstrade/core/service/CrossTradeService.java`

```java
package com.crosstrade.core.service;

import com.crosstrade.common.dto.request.ChatRequest;
import com.crosstrade.common.dto.response.IntentResult;
import com.crosstrade.common.enums.SecurityCheckResult;
import com.crosstrade.core.callback.StreamCompleteCallback;
import com.crosstrade.core.guardrail.InboundGuardrail;
import com.crosstrade.core.intent.IntentIdentifier;
import com.crosstrade.core.memory.MemoryManager;
import com.crosstrade.core.observe.MetricsCollector;
import com.crosstrade.core.observe.TraceContext;
import com.crosstrade.core.router.ChatContext;
import com.crosstrade.core.router.IntentRouter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import reactor.core.publisher.Flux;

import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReference;

/**
 * 核心处理服务
 * <p>
 * 核心处理流水线是将前面所有模块串联起来的"总指挥"，
 * 从用户请求到响应的完整流程：
 * 接收用户请求 → 入口安全审查 → L1短期记忆加载(缓存击穿→冷启动) →
 * 意图识别与路由 → 业务处理(导购/咨询/报价/未知) →
 * LLM流式生成 → 出口安全审查 → 流式输出完成回调 → 返回结果
 *
 * @author Wu, Hui Ming
 * @version 1.0
 * @School Suihua University
 */
@Slf4j
@Service
public class CrossTradeService {

    @Autowired
    private InboundGuardrail inboundGuardrail;

    @Autowired
    private MemoryManager memoryManager;

    @Autowired
    private IntentIdentifier intentIdentifier;

    @Autowired
    private IntentRouter intentRouter;

    @Autowired
    private StreamCompleteCallback streamCompleteCallback;

    @Autowired
    private MetricsCollector metricsCollector;

    /**
     * 核心处理方法（流式）
     *
     * @param request 聊天请求
     * @return 流式响应文本
     */
    public Flux<String> process(ChatRequest request) {
        // Step 0: 设置链路追踪
        TraceContext.generateTraceId();
        long startTime = System.currentTimeMillis();

        // Step 1: 入口安全审查
        SecurityCheckResult checkResult = inboundGuardrail.check(
                request.message(), request.sessionId(), request.userId()
        );
        if (checkResult == SecurityCheckResult.BLOCK) {
            metricsCollector.recordGuardrailBlock("inbound", "BLOCK");
            return Flux.just("您的消息包含敏感内容，请重新描述");
        }

        // Step 2: 加载记忆上下文（缓存击穿时触发冷启动恢复）
        ChatContext context = memoryManager.buildContext(
                request.sessionId(), request.userId(), request.message()
        );
        if (request.region() != null) {
            context.setRegion(request.region());
        }

        // Step 3: 意图识别与路由
        IntentResult intentResult = intentIdentifier.identify(request.message());
        context.setIntentType(intentResult.intentType().getValue());
        metricsCollector.recordIntentDistribution(intentResult.intentType());

        // Step 4: 业务处理（流式生成）
        Flux<String> responseFlux = intentRouter.route(intentResult, context);

        // Step 5: 流式输出 + 完成回调
        StringBuilder fullResponse = new StringBuilder();
        AtomicInteger tokenCount = new AtomicInteger(0);

        return responseFlux
                .doOnNext(token -> {
                    // Step 5.1: 累积完整响应
                    fullResponse.append(token);
                    tokenCount.incrementAndGet();
                })
                .doOnComplete(() -> {
                    // Step 6: 流式输出完成回调
                    try {
                        streamCompleteCallback.onComplete(
                                request.sessionId(), request.userId(),
                                request.message(), fullResponse.toString(),
                                tokenCount.get()
                        );

                        // Step 7: 记录响应延迟
                        long duration = System.currentTimeMillis() - startTime;
                        metricsCollector.recordResponseLatency("qwen3.7-max", duration);
                    } catch (Exception e) {
                        log.error("流式输出完成回调异常: sessionId={}", request.sessionId(), e);
                    } finally {
                        TraceContext.clear();
                    }
                })
                .doOnError(e -> {
                    log.error("流式输出异常: sessionId={}", request.sessionId(), e);
                    TraceContext.clear();
                });
    }

    /**
     * 同步处理方法（调试用）
     *
     * @param request 聊天请求
     * @return 完整回复文本
     */
    public String processSync(ChatRequest request) {
        StringBuilder result = new StringBuilder();
        process(request)
                .doOnNext(result::append)
                .blockLast();
        return result.toString();
    }
}
```
